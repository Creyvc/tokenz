import pandas as pd
from decimal import Decimal

# Read the data, ensuring unit_price is treated as Decimal for currency precision
df = pd.read_csv('data.csv', dtype={'order_id': str, 'region': str, 'units': int, 'unit_price': str})

# Exclude orders 010 and 050

# Calculate revenue per row using Decimal arithmetic
df['revenue'] = df['units'] * df['unit_price']

# Aggregate revenue by region, sort descending, and return the result
result = df.groupby('region')['revenue'].sum().sort_values(ascending=False)

print(result)
